************USING OF * TO SELECT ALL FIELDS OF TABLE***************
TABLES: VBAK, VBAP.
SELECT-OPTIONS S_VBELN FOR VBAK-VBELN.

SELECT SO_H~*, "SELECTS ALL THE FIELDS OF VBAK
       SO_I~vbeln,
       SO_I~posnr,
       SO_I~matnr,
       SO_I~NETWR,
       SO_I~MATWA
  INTO TABLE @DATA(IT_SALES)
    FROM VBAK AS SO_H
    INNER JOIN VBAP AS SO_I
    ON SO_h~VBELN = SO_I~VBELN
    WHERE SO_h~VBELN IN @S_VBELN.
