define view ZDWIJA_VIEW_3
with parameters P_VBELNL : vbeln_va,
                P_VBELNH : vbeln_va

as select from vbap as TAB
{
key TAB.vbeln as TAB_ORDER,     //USE OF ALIASES
key TAB.posnr,
    TAB.matnr as MATERIAL_NO,
    TAB.matkl,
    TAB.matwa,
    TAB.netpr,
    TAB.netwr
} where TAB.vbeln between :P_VBELNL and :P_VBELNH;
