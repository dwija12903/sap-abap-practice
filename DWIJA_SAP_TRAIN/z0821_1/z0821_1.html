*&---------------------------------------------------------------------*

*&---------------------------------------------------------------------*

*&---------------------------------------------------------------------*
REPORT Z0821_1.

types: begin of ls_tab,
         emplid       type z0724_emplid,
         empl_name    type z0724_empl_name,
         empl_age     type z0724_empl_age,
         empl_address type z0724_empl_address,
         empl_contact type z0724_empl_contact,
       end of ls_tab.

data: lt_tab type STANDARD TABLE OF ls_tab,
      wa_tab type ls_tab,
      lv_emplid type z0724_emplid.

*enter range
select-options s_emplid for lv_emplid.

*choose radio button
*add radio button name from selection text
selection-screen begin of block b1 with frame title text-t01 .
  parameters:rad1 radiobutton group rad user-command flag default 'X',
             rad2 radiobutton group rad .
selection-screen end of block b1.

selection-screen begin of block b2 with frame title text-t02 .
  parameters: p_age type z0724_empl_age modif id m1,
              p_name type z0724_empl_name modif id m2.
selection-screen end of block b2.

at selection-screen output.
loop at screen.
  if rad1 = 'X' and
  screen-group1 = 'M2'.
    screen-active = '0'. "M2 will be deactivated.

elseif rad2 = 'X' and
  screen-group1 = 'M1'.
    screen-active = '0'.
  endif.
  modify screen.
endloop.

*execution
start-of-selection.
if rad1 = 'X'.
  perform get_age.
else.
  perform get_name.
endif.

form get_age .

  select emplid empl_name empl_age empl_address empl_contact
    into table lt_tab
    from z0724_tb_1_empl
    where emplid in s_emplid
     and empl_age eq p_age .

*output to display.
  loop at lt_tab into wa_tab.
    write : / wa_tab-emplid,
    wa_tab-empl_name,
    wa_tab-empl_age,
    wa_tab-empl_address.
  endloop.

endform.

form get_name .
   select emplid empl_name empl_age empl_address empl_contact
    into table lt_tab
    from z0724_tb_1_empl
    where emplid in s_emplid
     and empl_age eq p_name .

  loop at lt_tab into wa_tab.
    write : / wa_tab-emplid,
    wa_tab-empl_name,
    wa_tab-empl_age,
    wa_tab-empl_address.
  endloop.

endform.
