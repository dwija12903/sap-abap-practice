report z0730_2 no standard page heading message-id zdwija_msg line-count 25(3).

include z0730_2_incl. "Declaration of constant parameters.

select-options s_matnr for lv_matnr.

initialization. "event decalaration of variables.
  perform sr1.

at selection-screen on s_matnr. "event for input validation.
  perform sr2.

  perform sr3.

end-of-selection. "event for output screen display.
  perform sr6.

  perform sr4.

end-of-page. "event for footer.
  perform sr5.

*===============Sub-routines==================
form sr1 .

  s_matnr-high = '100'.
  s_matnr-low = '1'.
  append s_matnr.

endform.


form sr2 .

  select single matnr into lv_matnr
      from mara
      where matnr in s_matnr.

  if sy-subrc eq 0.
    message s000.
*      MESSAGE I002.
  else.
*      MESSAGE E001.
*      MESSAGE W003.
*      MESSAGE A004.
    message x005.
  endif.

endform.

form sr3 .
  select matnr mbrsh mtart matkl meins brgew ntgew
  into table lt_tab
  from mara
  where matnr in s_matnr.
endform.

form sr4 .

  uline.
  write:/1 sy-vline, 'Material Number',
    30 sy-vline,'Industry Sector',
    50 sy-vline,'Mat Type',
    70 sy-vline,'mat grp',
    90 sy-vline,'base unit of measure',
    110 sy-vline,'gross weight',
    130 sy-vline,'net weight'.
  uline.
endform.

form sr5 .
  write: 'page.no',sy-pagno, 'date',sy-datum, 'time',sy-uzeit.
endform.

form sr6 .
  loop at lt_tab into wa_tab.
    write: /1 sy-vline, wa_tab-matnr,
    30 sy-vline, wa_tab-mbrsh,
    50 sy-vline, wa_tab-mtart,
    70 sy-vline,wa_tab-matkl,
    90 sy-vline, wa_tab-meins,
    110 sy-vline, wa_tab-brgew,
    130 sy-vline, wa_tab-ntgew.
    uline.
  endloop.


  data lv_lines type i.
  lv_lines = sy-linct - sy-linno.
  skip lv_lines.
endform.
