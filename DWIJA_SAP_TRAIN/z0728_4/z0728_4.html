Description: Internal Tables with Joins
*&---------------------------------------------------------------------*
*& Report Z0728_4
*&---------------------------------------------------------------------*
*&
*&---------------------------------------------------------------------*
REPORT Z0728_4 NO STANDARD PAGE HEADING.

include Z0728_4_INCL.

select-options ss_id for s_id.

*selecting fields of both tables using ~
select tb1~emplid tb1~empl_name tb1~empl_age tb1~empl_address tb1~empl_contact
  tb2~dept_no tb2~dept_name

  into table it_tab                   "copying into Interal Table.
  "joining 2 tables.
  from Z0724_TB_1_EMPL as tb1
  INNER JOIN Z0725_TB_1_DEPT as tb2
  on tb1~emplid = tb2~empl_id

  where tb1~emplid in ss_id.

sort it_tab by emplid ASCENDING.

if sy-subrc eq 0.
    write :/ 'records are found in mara table', sy-dbcnt.
    uline.
  else.
    write :/ 'records are not found in mara table', sy-dbcnt.
    uline.
endif.

skip 2.

Write: /3 'Row Number',
       5 'Id',
       15 'Name',
       30 'age',
       45 'Address',
       50 'contact No.',
       60 'Dept NO',
       70 'Dept Name'.

LOOP at it_tab into wa_tab.
  write:/3 sy-tabix,
  5 wa_tab-emplid,
  15 wa_tab-empl_name,
  30 wa_tab-empl_age,
  45 wa_tab-empl_address,
  50 wa_tab-empl_contact,
  60 wa_tab-dept_no,
  70 wa_tab-dept_name.
endloop.
