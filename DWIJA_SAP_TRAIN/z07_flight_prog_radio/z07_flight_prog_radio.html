Description: This is the program with radio and other fields

REPORT Z07_FLIGHT_PROG_RADIO.

tables sflight.




select-options date for sflight-fldate.

SELECTION-SCREEN skip 1.

PARAMETERS : economy RADIOBUTTON GROUP rg,
business RADIOBUTTON GROUP rg,
first RADIOBUTTON GROUP rg.

data : int_tab type table of Z7895_SFLIGHT_STR,
       wa type Z7895_SFLIGHT_STR.

select CARRID CONNID FLDATE SEATSMAX SEATSOCC SEATSMAX_B SEATSOCC_B SEATSMAX_F SEATSOCC_F into CORRESPONDING FIELDS OF table int_tab
  from sflight where fldate between date-low and date-high.




if economy = 'X'.
  loop at int_tab into wa.
  write : / wa-carrid, wa-connid, wa-fldate, wa-seatsmax, wa-seatsocc.
endloop.
elseif business = 'X'.
  loop at int_tab into wa.
  write : / wa-carrid, wa-connid, wa-fldate, wa-seatsmax_b, wa-seatsocc_b.
endloop.
elseif first = 'X'.
  loop at int_tab into wa.
  write : / wa-carrid, wa-connid, wa-fldate, wa-seatsmax_f, wa-seatsocc_f.
endloop.
endif.
