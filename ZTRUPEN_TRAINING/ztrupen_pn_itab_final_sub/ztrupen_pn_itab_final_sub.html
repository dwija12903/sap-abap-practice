report ztrupen_pn_itab_final_sub NO STANDARD PAGE HEADING.

include ZTRUPEN_PN_ITAB_FINAL_SUB_TOP.

select-options s_matnr for lv_matnr.
start-of-selection.

PERFORM GET_MAT_DATA.

PERFORM DISP_MAT_DATA.

WRITE 'CAPGEMINI'.


*&---------------------------------------------------------------------*
*&      Form  GET_MAT_DATA
*&---------------------------------------------------------------------*
FORM get_mat_data .


  select matnr mbrsh mtart matkl meins brgew ntgew
    into table lt_tab
       from mara
       where matnr in s_matnr.

  sort lt_tab by matnr ascending.

****IF LT_TAB[] IS NOT INITIAL.
  if sy-subrc eq 0.
    write :/ 'Materaial Master records are found in mara table', sy-dbcnt.
    ULINE.
  else.
    write :/ 'Materaial Master records are not found in mara table', sy-dbcnt.
    ULINE.
  endif.

PERFORM GET_HEADER_INFO.



ENDFORM.


*&---------------------------------------------------------------------*
*&      Form  GET_HEADER_INFO
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
*  -->  p1        text
*  <--  p2        text
*----------------------------------------------------------------------*
FORM get_header_info .

skip 2.

WRITE:/3 'Row Numb',
       15 'Mat Number',
       25 'Mat Ind',
       40 'Mat Type',
       56 'Mat grp',
       64 'Mat UOM',
       80 'Mat Gross wt',
       100 'Mat Net wt'.
ULINE.



ENDFORM.
*&---------------------------------------------------------------------*
*&      Form  DISP_MAT_DATA
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
*  -->  p1        text
*  <--  p2        text
*----------------------------------------------------------------------*
FORM disp_mat_data .

  loop at lt_tab into wa_tab.
    write :/3 sy-tabix,
    15 wa_tab-matnr,
    25 wa_tab-mbrsh,
    40 wa_tab-mtart,
    56 wa_tab-matkl,
    64 wa_tab-meins,
    80 wa_tab-brgew,
    100 wa_tab-ntgew.

  endloop.
ENDFORM.
