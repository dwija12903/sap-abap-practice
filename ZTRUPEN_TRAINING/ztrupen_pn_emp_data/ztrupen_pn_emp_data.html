REPORT ztrupen_pn_emp_data.

TABLES ztrupn_pn_emp.

DATA: lv_empno TYPE ztrupen_pn_empno, " EMP NO
      lv_empname  TYPE ZTRUPEN_PN_EMPNAME, " EMP NAME
      lv_empsal   TYPE  ztrupen_pn_empsal, " EMP SAL
      lv_empaddrs TYPE ztrupen_pn_empaddrs, " EMPADDRS
      lv_city     TYPE  zcity.  " City


SELECT-OPTIONS s_empno FOR lv_empno.

START-OF-SELECTION.

  SELECT empno empname empsal empaddrs city
         INTO ( lv_empno, lv_empname, lv_empsal, lv_empaddrs, lv_city )
    FROM ztrupn_pn_emp
     WHERE empno IN s_empno.

    WRITE :/ lv_empno, lv_empname, lv_empsal, lv_empaddrs, lv_city.
    CLEAR : lv_empno, lv_empname, lv_empsal, lv_empaddrs, lv_city.
  ENDSELECT.

  IF sy-subrc EQ 0.
    WRITE :/ 'Emp Records are found' , sy-dbcnt.
  ELSE.
    WRITE :/ 'Emp Records are not found' , sy-dbcnt.
  ENDIF.
