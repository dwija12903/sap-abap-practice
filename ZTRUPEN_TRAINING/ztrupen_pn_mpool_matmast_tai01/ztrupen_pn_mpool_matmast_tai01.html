MODULE USER_COMMAND_0100 INPUT.
DATA OK_CODE TYPE SY-UCOMM.
*  CASE  SY-UCOMM.
CASE OK_CODE.
    WHEN 'GETMAT'.
      SELECT MATNR MBRSH MTART MEINS BRGEW NTGEW
     INTO TABLE LT_TAB
        FROM MARA
        WHERE MATNR BETWEEN LV_MATNR_L AND LV_MATNR_H.
      LEAVE TO SCREEN 0200.

    WHEN 'EXIT'.
      LEAVE TO SCREEN 0.

  ENDCASE.



ENDMODULE.
