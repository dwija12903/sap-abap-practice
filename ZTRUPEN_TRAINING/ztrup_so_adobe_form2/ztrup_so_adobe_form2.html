REPORT ztrup_so_adobe_form2.

DATA: gv_vbeln TYPE vbak-vbeln,
      gv_func_name  TYPE funcname,
      gs_output     TYPE ssfctrlop,
      gs_customer   TYPE ztrup_customer,
      gs_outp_param TYPE sfpoutputparams,
      gt_final      TYPE ztrup_t_vbak2.

SELECT-OPTIONS:s_vbeln FOR gv_vbeln OBLIGATORY.

SELECTION-SCREEN: BEGIN OF BLOCK b1 WITH FRAME TITLE TEXT-000.
PARAMETERS: p_adobe RADIOBUTTON GROUP form,
            p_smart RADIOBUTTON GROUP form.
SELECTION-SCREEN: END OF BLOCK b1.

AT SELECTION-SCREEN.
* -- Validate Sales orders

  SELECT SINGLE vbeln
  FROM vbak
  INTO gv_vbeln
  WHERE vbeln IN s_vbeln.

  IF sy-subrc IS NOT INITIAL.
    MESSAGE e017(pord_messages).
  ENDIF.

START-OF-SELECTION.
  SELECT vbeln, vbak~erdat, vbtyp, vbak~kunnr, land1, adrnr
    FROM vbak
    INNER JOIN kna1 ON kna1~kunnr = vbak~kunnr
    INTO TABLE @DATA(gt_vbak)
    WHERE vbeln IN @s_vbeln.
  IF sy-subrc IS INITIAL.
    SELECT vbeln, posnr, matnr, kwmeng, vrkme
      FROM vbap
      INTO TABLE @DATA(gt_vbap)
      FOR ALL ENTRIES IN @gt_vbak
      WHERE vbeln = @gt_vbak-vbeln.
    IF sy-subrc IS NOT INITIAL.
      CLEAR: gt_vbak, gt_vbap.
      MESSAGE e017(prod_messages).
    ENDIF.
  ENDIF.

  IF gt_vbak IS NOT INITIAL.
* -- Parallel Cursor
* -- Pre-requisite to sort with common fields used in Parallel cursor
    SORT gt_vbak BY vbeln.
    SORT gt_vbap BY vbeln.
    DATA(gv_index) = VALUE syst_tabix( ).
    LOOP AT gt_vbak ASSIGNING FIELD-SYMBOL(<lfs_vbak>).
      APPEND INITIAL LINE TO gt_final ASSIGNING FIELD-SYMBOL(<gfs_final>).
      MOVE-CORRESPONDING <lfs_vbak> TO <gfs_final>.

      READ TABLE gt_vbap TRANSPORTING NO FIELDS
      WITH KEY vbeln = <lfs_vbak>-vbeln BINARY SEARCH.
      IF sy-subrc IS INITIAL.
        gv_index = sy-tabix.
        LOOP AT gt_vbap ASSIGNING FIELD-SYMBOL(<lfs_vbap>) FROM gv_index.
          IF <lfs_vbap>-vbeln NE <lfs_vbak>-vbeln.
            EXIT.
          ENDIF.
          APPEND INITIAL LINE TO <gfs_final>-item ASSIGNING FIELD-SYMBOL(<lfs_items>).
          MOVE-CORRESPONDING <lfs_vbap> TO <lfs_items>.
        ENDLOOP.
      ENDIF.
    ENDLOOP.
    IF p_adobe EQ abap_true.
      TRY.
          CALL FUNCTION 'FP_FUNCTION_MODULE_NAME'
            EXPORTING
              i_name     = 'ZTRUP_SALES_ORDER_FORM2'
            IMPORTING
              e_funcname = gv_func_name.
        CATCH cx_root.
      ENDTRY.
    ELSE.
      CALL FUNCTION 'SSF_FUNCTION_MODULE_NAME'
        EXPORTING
          formname           = 'ZTRUP_SALES_ORDER_FORM2'
        IMPORTING
          fm_name            = gv_func_name
        EXCEPTIONS
          no_form            = 1
          no_function_module = 2
          OTHERS             = 3.
      IF sy-subrc <> 0.
* Implement suitable error handling here
      ENDIF.
    ENDIF.

    IF gv_func_name IS NOT INITIAL.
      gs_outp_param-device = 'LOCL'.
      gs_outp_param-nodialog = abap_true.
      gs_outp_param-preview = abap_true.

      CALL FUNCTION gv_func_name
        EXPORTING
          iv_logo          = 'SAPLOGO'
          iv_terms         = 'ZTRUP_SALES_ORDER_COND'
          it_data          = gt_final
        EXCEPTIONS
          formatting_error = 1
          internal_error   = 2
          send_error       = 3
          user_canceled    = 4
          logo_fail        = 5
          OTHERS           = 6.
      IF sy-subrc IS INITIAL.

      ENDIF.

    ENDIF.
  ENDIF.
