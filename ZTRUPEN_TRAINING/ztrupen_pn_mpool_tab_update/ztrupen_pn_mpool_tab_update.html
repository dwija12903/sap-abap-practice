PROGRAM ZTRUPEN_PN_MPOOL_TAB_UPDATE.

TABLES: ZTRUPN_PN_EMP.

TYPES: BEGIN OF LS_EMP,
         MANDT    TYPE MANDT,
         EMPNO    TYPE ZTRUPEN_PN_EMPNO,
         EMPNAME  TYPE ZTRUPEN_PN_EMPNAME,
         EMPSAL   TYPE ZTRUPEN_PN_EMPSAL,
         EMPADDRS TYPE ZTRUPEN_PN_EMPADDRS,
         CITY     TYPE ZCG_CITY,

       END OF LS_EMP.

DATA : LT_TAB TYPE STANDARD TABLE OF LS_EMP,
       WA_TAB TYPE LS_EMP.



*&---------------------------------------------------------------------*
*&      Module  STATUS_0100  OUTPUT
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
MODULE STATUS_0100 OUTPUT.
  SET PF-STATUS SPACE.
*  SET TITLEBAR 'xxx'.
ENDMODULE.
*&---------------------------------------------------------------------*
*&      Module  HIDE_OPT  OUTPUT
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
MODULE HIDE_OPT OUTPUT.
  LOOP AT SCREEN.
    CASE SY-UCOMM.
      WHEN 'INSERT'.  " G1
        IF SCREEN-GROUP1 EQ 'G2' OR SCREEN-GROUP1 EQ 'G3'.
          SCREEN-INPUT = 0.
          MODIFY SCREEN.
        ENDIF.

      WHEN 'UPDATE'.  " G2
        IF SCREEN-GROUP1 EQ 'G1' OR SCREEN-GROUP1 EQ 'G3'.
          SCREEN-INPUT = 0.
          MODIFY SCREEN.
        ENDIF.

      WHEN 'DELETE'.  " G3
        IF SCREEN-GROUP1 EQ 'G1' OR SCREEN-GROUP1 EQ 'G2'.
          SCREEN-INPUT = 0.
          MODIFY SCREEN.
        ENDIF.

      WHEN 'CLEAR'.
        IF SCREEN-GROUP1 EQ 'G1' OR SCREEN-GROUP1 EQ 'G2' OR SCREEN-GROUP1 EQ 'G3'.
          SCREEN-INPUT = 1.
          MODIFY SCREEN.
        ENDIF.
    ENDCASE.
  ENDLOOP.

ENDMODULE.
*&---------------------------------------------------------------------*
*&      Module  USER_COMMAND_0100  INPUT
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
MODULE USER_COMMAND_0100 INPUT.

  CASE SY-UCOMM .
      WHEN 'INSERT'.
      PERFORM INSERT_DATA.
      WHEN 'UPDATE'.
      PERFORM UPDATE_DATA.
    WHEN 'DELETE'.
      PERFORM DELETE_DATA.
    WHEN 'DISP'.
      PERFORM DISP_DATA.
    WHEN  'CLEAR'.


CLEAR : ZTRUPN_PN_EMP-EMPNO, ZTRUPN_PN_EMP-EMPNAME,
ZTRUPN_PN_EMP-EMPSAL, ZTRUPN_PN_EMP-EMPADDRS, ZTRUPN_PN_EMP-CITY.

WHEN 'EXIT'.
        LEAVE TO SCREEN 0.
  ENDCASE.



ENDMODULE.
*&---------------------------------------------------------------------*
*&      Form  INSERT_DATA
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
*  -->  p1        text
*  <--  p2        text
*----------------------------------------------------------------------*
FORM INSERT_DATA .
MOVE ZTRUPN_PN_EMP TO WA_TAB.

APPEND WA_TAB TO LT_TAB.
INSERT INTO ZTRUPN_PN_EMP VALUES WA_TAB.

COMMIT WORK .

IF SY-SUBRC EQ 0.
  MESSAGE I000(ZTRUPEN_EMP).
  ELSE.
    MESSAGE E004(ZTRUPEN_EMP).
  ENDIF.

ENDFORM.
*&---------------------------------------------------------------------*
*&      Form  UPDATE_DATA
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
*  -->  p1        text
*  <--  p2        text
*----------------------------------------------------------------------*
FORM UPDATE_DATA .

MOVE ZTRUPN_PN_EMP TO WA_TAB.

APPEND WA_TAB TO LT_TAB.
UPDATE ZTRUPN_PN_EMP FROM TABLE LT_TAB.
COMMIT WORK .

IF SY-SUBRC EQ 0.
  MESSAGE I001(ZTRUPEN_EMP).
  ELSE.
    MESSAGE E004(ZTRUPEN_EMP).
  ENDIF.

ENDFORM.
*&---------------------------------------------------------------------*
*&      Form  DELETE_DATA
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
*  -->  p1        text
*  <--  p2        text
*----------------------------------------------------------------------*
FORM DELETE_DATA .
MOVE ZTRUPN_PN_EMP TO WA_TAB.

APPEND WA_TAB TO LT_TAB.
DELETE FROM ZTRUPN_PN_EMP WHERE EMPNO = WA_TAB-EMPNO.


IF SY-SUBRC EQ 0.
  MESSAGE I002(ZTRUPEN_EMP).
  ELSE.
    MESSAGE E004(ZTRUPEN_EMP).
  ENDIF.
ENDFORM.
*&---------------------------------------------------------------------*
*&      Form  DISP_DATA
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
*  -->  p1        text
*  <--  p2        text
*----------------------------------------------------------------------*
FORM DISP_DATA .
SELECT SINGLE EMPNO
EMPNAME
EMPSAL
EMPADDRS
CITY
  INTO ( ZTRUPN_PN_EMP-EMPNO, ZTRUPN_PN_EMP-EMPNAME, ZTRUPN_PN_EMP-EMPSAL,
  ZTRUPN_PN_EMP-EMPADDRS, ZTRUPN_PN_EMP-CITY )
  FROM ZTRUPN_PN_EMP
  WHERE EMPNO EQ ZTRUPN_PN_EMP-EMPNO.
ENDFORM.
